cmake_minimum_required(VERSION 3.24)
project(EdgeTuck VERSION 0.8.10 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "EdgeTuck currently requires Windows and the MSVC C++ toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(edgetuck_core src/layout.cpp src/model.cpp src/references.cpp src/sorting.cpp)
target_include_directories(edgetuck_core PUBLIC src)
target_compile_definitions(edgetuck_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(edgetuck_core PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(edgetuck_core PUBLIC shell32 ole32 shlwapi)

get_filename_component(WINDOWS_KITS_ROOT "[HKEY_LOCAL_MACHINE\\SOFTWARE\\Microsoft\\Windows Kits\\Installed Roots;KitsRoot10]" ABSOLUTE)
file(GLOB SDK_FXC "${WINDOWS_KITS_ROOT}/bin/*/x64/fxc.exe")
file(GLOB SDK_HLSL_HELPER "${WINDOWS_KITS_ROOT}/Include/*/um/d2d1effecthelpers.hlsli")
list(SORT SDK_HLSL_HELPER COMPARE NATURAL ORDER DESCENDING)
list(GET SDK_HLSL_HELPER 0 SDK_HLSL_HELPER)
get_filename_component(SDK_HLSL_INCLUDE "${SDK_HLSL_HELPER}" DIRECTORY)
list(SORT SDK_FXC COMPARE NATURAL ORDER DESCENDING)
find_program(FXC_EXECUTABLE fxc HINTS ${SDK_FXC})
if(NOT FXC_EXECUTABLE AND SDK_FXC)
  list(GET SDK_FXC 0 FXC_EXECUTABLE)
endif()
if(NOT FXC_EXECUTABLE)
  message(FATAL_ERROR "Windows SDK HLSL compiler (fxc.exe) is required.")
endif()
set(SHADER_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/optical_shader.h")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${SHADER_HEADER}"
  COMMAND "${FXC_EXECUTABLE}" /nologo /T ps_4_0 /E main /O3 /Ges /I "${SDK_HLSL_INCLUDE}" /Fh "${SHADER_HEADER}" /Vn edge_optical_shader "${CMAKE_CURRENT_SOURCE_DIR}/src/optical.hlsl"
  DEPENDS src/optical.hlsl VERBATIM)
add_executable(EdgeTuck WIN32 src/main.cpp src/app.cpp src/diagnostics.cpp src/live_test.cpp src/desktop_grid.cpp src/desktop_capture.cpp src/file_images.cpp src/material_window.cpp src/drawer.cpp src/file_interaction.cpp src/shell_files.cpp src/paint.cpp src/composition.cpp src/optical_effect.cpp src/drop_target.cpp resources/app.rc "${SHADER_HEADER}")
target_sources(EdgeTuck PRIVATE src/archive.cpp src/archive_app.cpp src/desktop_icons.cpp src/archive_test.cpp src/runtime_options.cpp src/storage.cpp src/storage_move.cpp src/storage_window.cpp src/menu_host.cpp src/drawer_sort.cpp)
target_sources(EdgeTuck PRIVATE src/explorer_entry.cpp)
target_sources(EdgeTuck PRIVATE src/archive_rename.cpp)
target_sources(EdgeTuck PRIVATE src/control_window.cpp)
target_compile_options(EdgeTuck PRIVATE /W4 /utf-8 /permissive-)
target_include_directories(EdgeTuck PRIVATE resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(EdgeTuck PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(EdgeTuck PRIVATE edgetuck_core d2d1 dwrite dwmapi windowscodecs comctl32 shlwapi uxtheme psapi d3d11 dxgi dxguid windowsapp CoreMessaging oleaut32 bcrypt)
target_link_options(EdgeTuck PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest)
